cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vibration_serial_bridge LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  add_subdirectory(../core ${CMAKE_CURRENT_BINARY_DIR}/core)
endif()
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "The board serial bridge requires Linux nonblocking file descriptors")
endif()
find_package(Qt5 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
add_executable(vibration_serial_bridge
  main.cpp
  ../workbench/capture_client.cpp ../workbench/capture_client.hpp
  ../workbench/capture_transport.cpp ../workbench/capture_transport.hpp
  ../workbench/live_preview_model.cpp ../workbench/live_preview_model.hpp)
target_include_directories(vibration_serial_bridge PRIVATE ../workbench)
target_link_libraries(vibration_serial_bridge PRIVATE vibration_core Qt5::Core Qt5::Widgets)
target_compile_features(vibration_serial_bridge PRIVATE cxx_std_17)
target_compile_options(vibration_serial_bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS vibration_serial_bridge RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(vibration_raw_window_stats_test tests/raw_window_stats_test.cpp)
    target_include_directories(vibration_raw_window_stats_test PRIVATE . ../workbench)
    target_link_libraries(vibration_raw_window_stats_test PRIVATE Qt5::Widgets)
    target_compile_features(vibration_raw_window_stats_test PRIVATE cxx_std_17)
    add_test(NAME raw_window_amplitude_stats COMMAND vibration_raw_window_stats_test)
endif()
