add_library(vibration_core src/core.cpp src/detection.cpp src/candidate_model.cpp src/reference_library.cpp src/reference_validation.cpp src/supervised_features.cpp src/supervised_model.cpp)
find_package(Threads REQUIRED)
target_link_libraries(vibration_core PUBLIC Threads::Threads vibration_accelerator)
target_include_directories(vibration_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(vibration_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(vibration_core PRIVATE /W4)
else()
  target_compile_options(vibration_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vibration_analyze src/cli.cpp)
add_executable(vibration_detect src/detect_cli.cpp)
target_link_libraries(vibration_detect PRIVATE vibration_core)
target_link_libraries(vibration_analyze PRIVATE vibration_core)
if(WIN32)
  target_link_libraries(vibration_analyze PRIVATE shell32)
endif()
if(MINGW)
  target_link_options(vibration_analyze PRIVATE -static -static-libgcc -static-libstdc++)
endif()
install(TARGETS vibration_core vibration_analyze vibration_detect)
install(DIRECTORY include/ DESTINATION include)
