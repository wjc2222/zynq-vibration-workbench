cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vibration_stream_bridge LANGUAGES CXX)
  include(CTest)
  add_subdirectory(../core ${CMAKE_CURRENT_BINARY_DIR}/core)
  add_subdirectory(../streaming ${CMAKE_CURRENT_BINARY_DIR}/streaming)
  target_include_directories(vibration_stream_service PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../hardware/combined/include)
endif()
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "The stream console bridge requires Linux file descriptors")
endif()
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(Threads REQUIRED)
add_executable(vibration_stream_bridge main.cpp records.hpp protocol.hpp display_payload.hpp)
target_include_directories(vibration_stream_bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../hardware/combined/include)
target_link_libraries(vibration_stream_bridge PRIVATE vibration_core Qt5::Core Threads::Threads)
target_compile_features(vibration_stream_bridge PRIVATE cxx_std_17)
target_compile_options(vibration_stream_bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS vibration_stream_bridge RUNTIME DESTINATION bin)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(vibration_stream_delete_labels_test tests/delete_labels_test.cpp ../workbench/record_labels.cpp)
  target_include_directories(vibration_stream_delete_labels_test PRIVATE .)
  target_compile_definitions(vibration_stream_delete_labels_test PRIVATE VIBRATION_STREAM_DELETE_TEST_HOOKS=1)
  target_compile_features(vibration_stream_delete_labels_test PRIVATE cxx_std_17)
  target_link_libraries(vibration_stream_delete_labels_test PRIVATE vibration_core Qt5::Core)
  add_test(NAME stream_delete_operator_labels COMMAND vibration_stream_delete_labels_test)
  add_executable(vibration_stream_display_payload_test tests/display_payload_test.cpp)
  target_include_directories(vibration_stream_display_payload_test PRIVATE . ${CMAKE_CURRENT_SOURCE_DIR}/../../hardware/combined/include)
  target_compile_features(vibration_stream_display_payload_test PRIVATE cxx_std_17)
  target_link_libraries(vibration_stream_display_payload_test PRIVATE vibration_core Qt5::Core Threads::Threads)
  add_test(NAME stream_display_raw_payload COMMAND vibration_stream_display_payload_test)
  add_executable(vibration_stream_delete_race_test tests/delete_race_test.cpp)
  target_include_directories(vibration_stream_delete_race_test PRIVATE .)
  target_compile_definitions(vibration_stream_delete_race_test PRIVATE VIBRATION_STREAM_DELETE_TEST_HOOKS=1)
  target_compile_features(vibration_stream_delete_race_test PRIVATE cxx_std_17)
  target_link_libraries(vibration_stream_delete_race_test PRIVATE vibration_core Qt5::Core)
  add_test(NAME stream_delete_directory_identity COMMAND vibration_stream_delete_race_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND AND TARGET vibration_stream_service)
    add_test(NAME stream_bridge_protocol COMMAND ${Python3_EXECUTABLE} -B
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bridge.py
      --bridge $<TARGET_FILE:vibration_stream_bridge>
      --service $<TARGET_FILE:vibration_stream_service>)
    set_tests_properties(stream_bridge_protocol PROPERTIES TIMEOUT 120)
  endif()
endif()
