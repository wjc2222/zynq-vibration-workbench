cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vibration_accelerator LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  include(CTest)
endif()
add_library(vibration_accelerator src/accelerator.cpp)
target_include_directories(vibration_accelerator PUBLIC include PRIVATE ../../hardware/combined/include ../../hardware/analysis_v6/data)
target_compile_features(vibration_accelerator PUBLIC cxx_std_17)
add_executable(vibration-accel-cli src/cli.cpp)
target_link_libraries(vibration-accel-cli PRIVATE vibration_accelerator)
if(UNIX)
  find_package(Threads REQUIRED)
  add_executable(vibration-accel-driver-test tests/driver_negative_test.cpp)
  target_include_directories(vibration-accel-driver-test PRIVATE ../../hardware/combined/include)
  target_link_libraries(vibration-accel-driver-test PRIVATE Threads::Threads)
endif()
if(BUILD_TESTING)
  add_executable(accelerator_test tests/accelerator_test.cpp)
  target_link_libraries(accelerator_test PRIVATE vibration_accelerator)
  add_test(NAME accelerator_contract COMMAND accelerator_test)
  add_executable(accelerator_comparison_test tests/comparison_test.cpp)
  target_link_libraries(accelerator_comparison_test PRIVATE vibration_accelerator)
  add_test(NAME accelerator_comparison_faults COMMAND accelerator_comparison_test)
endif()
